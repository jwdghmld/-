#修改初始化条件，避免反复初始化
vim ~/.bashrc
# >>> conda initialize >>>
# 只在交互式Shell下才初始化conda
if [[ $- == *i* ]]; then
    __conda_setup="$('/home/miniconda3/bin/conda' 'shell.bash' 'hook' 2> /dev/null)"
    if [ $? -eq 0 ]; then
        eval "$__conda_setup"
    else
        if [ -f "/home/miniconda3/etc/profile.d/conda.sh" ]; then
            . "/home/miniconda3/etc/profile.d/conda.sh"
        else
            export PATH="/home/miniconda3/bin:$PATH"
        fi
    fi
    unset __conda_setup
else
    # 非交互式Shell只添加基本路径，不初始化
    export PATH="/home/miniconda3/bin:$PATH"
fi
# <<< conda initialize <<<

八、克隆节点。并对子节点修改静态ip并初始化集群
1，每个节点进行第一步：

2，设置集群的免密登录： 在dage用户下
ssh-keygen -t rsa
（一直回车）
发送密钥给其他节点：（每个节点都要执行一次，输入yes和密码）
ssh-copy-id node1
ssh-copy-id node2
ssh-copy-id node3

3, 关闭防火墙
systemctl stop firewalld
systemctl disable firewalld

4, 将准备好的配置文件直接放到对应的组件位置上spark，hadoop ，hive

5，初始化hadoop
hdfs namenode -format

6，配置node2的mysql（元数据库，hive和airflow的）
sudo dnf install -y mysql-server

开机自启动：
sudo systemctl start mysqld
sudo systemctl enable mysqld

配置utf8
sudo vim /etc/my.cnf.d/mysql-server.cnf

[mysqld]
   # ... 其他默认配置保持不变 ...
   character-set-server=utf8mb4
   collation-server=utf8mb4_general_ci
配置简单密码：
SET GLOBAL validate_password.policy = 0;
SET GLOBAL validate_password.length = 4;

修改root用户：
ALTER USER 'root'@'localhost' IDENTIFIED BY '123456';
创建远程连接的root用户：
CREATE USER 'root'@'%' IDENTIFIED  BY '123456';
GRANT ALL PRIVILEGES ON *.* TO 'root'@'%' WITH GRANT OPTION;
flush privileges;

7， 创建hive的元数据库：
CREATE DATABASE metastore DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;

把mysql的jar包放到 hive 的 lib 目录下,spark的jars目录下，

8，解决Guava冲突：hive与hadoop的
# 进入 hive 的 lib 目录删除低版本
rm -f $HIVE_HOME/lib/guava-19.0.jar

# 从 hadoop 拷贝高版本到 hive
cp $HADOOP_HOME/share/hadoop/common/lib/guava-27.0-jre.jar $HIVE_HOME/lib/

9，初始化hive
schematool -dbType mysql -initSchema

10，移除冲突的日志包（hadoop）
mv $HADOOP_HOME/share/hadoop/common/lib/slf4j-reload4j-1.7.36.jar $HADOOP_HOME/share/hadoop/common/lib/slf4j-reload4j-1.7.36.jar.bak
cp $HIVE_HOME/lib/log4j-slf4j-impl-2.17.1.jar $HADOOP_HOME/share/hadoop/common/lib/
cp $HIVE_HOME/lib/log4j-api-2.17.1.jar $HADOOP_HOME/share/hadoop/common/lib/
cp $HIVE_HOME/lib/log4j-core-2.17.1.jar $HADOOP_HOME/share/hadoop/common/lib/

给node2，3复制一份
scp $HIVE_HOME/lib/log4j-slf4j-impl-2.17.1.jar node2:$HADOOP_HOME/share/hadoop/common/lib/
scp $HIVE_HOME/lib/log4j-api-2.17.1.jar node2:$HADOOP_HOME/share/hadoop/common/lib/
scp $HIVE_HOME/lib/log4j-core-2.17.1.jar node2:$HADOOP_HOME/share/hadoop/common/lib/

scp $HIVE_HOME/lib/log4j-slf4j-impl-2.17.1.jar node3:$HADOOP_HOME/share/hadoop/common/lib/
scp $HIVE_HOME/lib/log4j-api-2.17.1.jar node3:$HADOOP_HOME/share/hadoop/common/lib/

九、配置airflow
1，准备好airflow的python环境
conda create -n airflow python=3.11 -y

2, 准备元数据库（mysql）用户使用的 root
CREATE DATABASE airflow CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

3，创建工作目录airflow
mkdir -p /opt/app/airflow
cd /opt/app/airflow
export AIRFLOW_HOME=/opt/app/airflow

4，安装airflow=2.10.5
av=2.10.5
pv=3.11
con="https://raw.githubusercontent.com/apache/airflow/constraints-${av}/constraints-${pv}.txt"
# 安装 Airflow，同时指定安装 MySQL 扩展包
pip install "apache-airflow==${av}" --constraint "${con}"
pip install PyMySQL

5，工作环境配置（第三步）
airflow config list > /dev/null

vim $AIRFLOW_HOME/airflow.cfg
编辑为mysql数据库
[database]
sql_alchemy_conn = mysql+pymysql://root:123456@node2:3306/airflow

配置执行器
[core]
executor = LocalExecutor

[core]
load_examples = False

修改时区
[core]
default_timezone = Asia/Shanghai

DAG文件存放位置 
[core]
